1   ArrayList

import java.util.ArrayList;
import java.util.List;

public class ArrayListDemo {
    public static void main(String[] args) {

        // --- Creation ---
        List<String> fruits = new ArrayList<>();

        // --- Adding elements ---
        fruits.add("Apple");            // [Apple]
        fruits.add("Banana");           // [Apple, Banana]
        fruits.add("Cherry");           // [Apple, Banana, Cherry]
        fruits.add(1, "Blueberry");    // Insert at index 1 → [Apple, Blueberry, Banana, Cherry]

        // --- Accessing ---
        String first = fruits.get(0);   // "Apple"  → O(1) random access
        int size = fruits.size();         // 4

        // --- Searching ---
        boolean has = fruits.contains("Banana");  // true  → O(n)
        int idx = fruits.indexOf("Cherry");       // 3

        // --- Updating ---
        fruits.set(0, "Avocado");       // Replace index 0 → [Avocado, Blueberry, Banana, Cherry]

        // --- Removing ---
        fruits.remove("Banana");         // By value → [Avocado, Blueberry, Cherry]
        fruits.remove(0);                // By index → [Blueberry, Cherry]

        // --- Iterating ---
        for (String f : fruits) {
            System.out.println(f);
        }

        // --- Sublist ---
        fruits.add("Date");
        fruits.add("Elderberry");
        List<String> sub = fruits.subList(1, 3);  // [Cherry, Date]  (fromIndex inclusive, toIndex exclusive)

        // --- Convert array ↔ list ---
        String[] arr = fruits.toArray(new String[0]);
        List<String> fromArr = List.of("X", "Y", "Z");  // Immutable

        // --- isEmpty / clear ---
        System.out.println(fruits.isEmpty());  // false
        fruits.clear();
        System.out.println(fruits.isEmpty());  // true
    }
}

2   LinkedList

import java.util.LinkedList;
import java.util.Deque;

public class LinkedListDemo {
    public static void main(String[] args) {

        LinkedList<String> list = new LinkedList<>();

        // --- List operations ---
        list.add("A");
        list.add("B");
        list.add("C");

        // --- Head / Tail operations (O(1)) ---
        list.addFirst("Z");     // [Z, A, B, C]
        list.addLast("D");      // [Z, A, B, C, D]

        String head = list.getFirst();  // "Z"
        String tail = list.getLast();   // "D"

        list.removeFirst();     // [A, B, C, D]
        list.removeLast();      // [A, B, C]

        // --- As a Queue (FIFO) ---
        list.offer("X");        // Add to tail → [A, B, C, X]
        String polled = list.poll();    // Remove from head → "A",  list = [B, C, X]
        String peeked = list.peek();    // Look at head → "B" (doesn't remove)

        // --- As a Stack (LIFO) ---
        list.push("TOP");       // Push to head → [TOP, B, C, X]
        String popped = list.pop();     // Pop from head → "TOP"

        System.out.println(list);  // [B, C, X]
    }
}

3   HashSet

import java.util.HashSet;
import java.util.Set;

public class HashSetDemo {
    public static void main(String[] args) {

        Set<String> colors = new HashSet<>();

        // --- Adding ---
        colors.add("Red");
        colors.add("Green");
        colors.add("Blue");
        colors.add("Red");         // Duplicate ignored! returns false

        System.out.println(colors);        // [Red, Blue, Green]  (no guaranteed order)
        System.out.println(colors.size());  // 3  (not 4)

        // --- Checking ---
        boolean has = colors.contains("Blue");  // true → O(1)

        // --- Removing ---
        colors.remove("Green");  // → O(1)

        // --- Iterating (order not guaranteed) ---
        for (String c : colors) {
            System.out.println(c);
        }

        // --- Set operations ---
        Set<String> set1 = new HashSet<>(Set.of("A", "B", "C"));
        Set<String> set2 = new HashSet<>(Set.of("B", "C", "D"));

        // Union
        Set<String> union = new HashSet<>(set1);
        union.addAll(set2);             // [A, B, C, D]

        // Intersection
        Set<String> inter = new HashSet<>(set1);
        inter.retainAll(set2);          // [B, C]

        // Difference
        Set<String> diff = new HashSet<>(set1);
        diff.removeAll(set2);           // [A]

        System.out.println("Union: " + union);
        System.out.println("Intersection: " + inter);
        System.out.println("Difference: " + diff);
    }
}

5   TreeSet

import java.util.TreeSet;
import java.util.NavigableSet;

public class TreeSetDemo {
    public static void main(String[] args) {

        TreeSet<Integer> nums = new TreeSet<>();

        nums.add(50);
        nums.add(20);
        nums.add(80);
        nums.add(10);
        nums.add(40);

        // Automatically sorted!
        System.out.println(nums);           // [10, 20, 40, 50, 80]

        // --- NavigableSet methods ---
        System.out.println(nums.first());   // 10  (smallest)
        System.out.println(nums.last());    // 80  (largest)

        System.out.println(nums.lower(40));    // 20  (strictly less than 40)
        System.out.println(nums.higher(40));   // 50  (strictly greater than 40)
        System.out.println(nums.floor(40));    // 40  (less than or equal)
        System.out.println(nums.ceiling(45));  // 50  (greater than or equal)

        // --- Range views ---
        System.out.println(nums.headSet(40));            // [10, 20]  (less than 40)
        System.out.println(nums.tailSet(40));            // [40, 50, 80]  (>= 40)
        System.out.println(nums.subSet(20, 80));         // [20, 40, 50]  (20 inclusive, 80 exclusive)

        // --- Descending ---
        System.out.println(nums.descendingSet());       // [80, 50, 40, 20, 10]

        // --- Poll (remove + return) ---
        System.out.println(nums.pollFirst());  // 10  (removes smallest)
        System.out.println(nums.pollLast());   // 80  (removes largest)
        System.out.println(nums);              // [20, 40, 50]

        // --- Custom Comparator (reverse order) ---
        TreeSet<String> words = new TreeSet<>((a, b) -> b.compareTo(a));
        words.add("Banana");
        words.add("Apple");
        words.add("Cherry");
        System.out.println(words);  // [Cherry, Banana, Apple]  (reverse alphabetical)
    }
}

6   PriorityQueue

import java.util.PriorityQueue;
import java.util.Collections;

public class PriorityQueueDemo {
    public static void main(String[] args) {

        // --- Min-Heap (default) ---
        PriorityQueue<Integer> minHeap = new PriorityQueue<>();
        minHeap.offer(30);
        minHeap.offer(10);
        minHeap.offer(20);

        System.out.println(minHeap.peek());   // 10  (smallest = highest priority)
        System.out.println(minHeap.poll());   // 10  (removes smallest)
        System.out.println(minHeap.poll());   // 20
        System.out.println(minHeap.poll());   // 30

        // --- Max-Heap ---
        PriorityQueue<Integer> maxHeap = new PriorityQueue<>(Collections.reverseOrder());
        maxHeap.offer(30);
        maxHeap.offer(10);
        maxHeap.offer(20);

        System.out.println(maxHeap.poll());   // 30  (largest first)
        System.out.println(maxHeap.poll());   // 20
        System.out.println(maxHeap.poll());   // 10

        // --- Custom objects ---
        PriorityQueue<String> tasks = new PriorityQueue<>((a, b) -> a.length() - b.length());
        tasks.offer("Send email");
        tasks.offer("Code");
        tasks.offer("Deploy to production");

        while (!tasks.isEmpty()) {
            System.out.println(tasks.poll());  // Code → Send email → Deploy to production  (shortest first)
        }
    }
}

7   ArrayDeque — Stack & Queue

import java.util.ArrayDeque;
import java.util.Deque;

public class ArrayDequeDemo {
    public static void main(String[] args) {

        // ========== AS A STACK (LIFO) ==========
        Deque<String> stack = new ArrayDeque<>();

        stack.push("First");     // bottom
        stack.push("Second");
        stack.push("Third");     // top

        System.out.println(stack.peek());   // "Third"  (top, doesn't remove)
        System.out.println(stack.pop());    // "Third"  (removes top)
        System.out.println(stack.pop());    // "Second"
        System.out.println(stack.pop());    // "First"


        // ========== AS A QUEUE (FIFO) ==========
        Deque<String> queue = new ArrayDeque<>();

        queue.offer("First");    // front
        queue.offer("Second");
        queue.offer("Third");    // back

        System.out.println(queue.peek());   // "First"  (front)
        System.out.println(queue.poll());   // "First"  (removes front)
        System.out.println(queue.poll());   // "Second"
        System.out.println(queue.poll());   // "Third"


        // ========== DOUBLE-ENDED ==========
        Deque<String> deque = new ArrayDeque<>();

        deque.offerFirst("B");    // [B]
        deque.offerFirst("A");    // [A, B]
        deque.offerLast("C");     // [A, B, C]

        System.out.println(deque.peekFirst());  // "A"
        System.out.println(deque.peekLast());   // "C"
        System.out.println(deque.pollFirst());  // "A"  → deque = [B, C]
        System.out.println(deque.pollLast());   // "C"  → deque = [B]
    }
}

💡 Always use ArrayDeque over Stack class

java.util.Stack extends Vector (synchronized, legacy, slow). ArrayDeque is the modern, faster replacement for both Stack and Queue use cases.

8   HashMap

import java.util.HashMap;
import java.util.Map;

public class HashMapDemo {
    public static void main(String[] args) {

        Map<String, Integer> scores = new HashMap<>();

        // --- put ---
        scores.put("Alice", 95);
        scores.put("Bob", 87);
        scores.put("Charlie", 92);
        scores.put("Alice", 98);   // Overwrites! Key "Alice" now → 98

        // --- get ---
        int aliceScore = scores.get("Alice");           // 98
        Integer unknown = scores.get("Zara");          // null (key doesn't exist)
        int safe = scores.getOrDefault("Zara", 0);      // 0 (default if missing)

        // --- Check existence ---
        scores.containsKey("Bob");       // true
        scores.containsValue(87);       // true

        // --- Remove ---
        scores.remove("Charlie");        // removes the entry

        // --- Iterate over entries ---
        for (Map.Entry<String, Integer> entry : scores.entrySet()) {
            System.out.println(entry.getKey() + " = " + entry.getValue());
        }

        // --- Iterate keys only ---
        for (String key : scores.keySet()) {
            System.out.println(key);
        }

        // --- Iterate values only ---
        for (int val : scores.values()) {
            System.out.println(val);
        }

        // --- putIfAbsent (only put if key doesn't exist) ---
        scores.putIfAbsent("Alice", 100);   // Does nothing, Alice already exists
        scores.putIfAbsent("Diana", 88);   // Added!

        // --- compute / merge (great for counting) ---
        Map<String, Integer> wordCount = new HashMap<>();
        String[] words = {"hello", "world", "hello", "java", "hello"};

        for (String w : words) {
            wordCount.merge(w, 1, Integer::sum);  // If exists, add 1; else put 1
        }
        System.out.println(wordCount);  // {java=1, world=1, hello=3}

        // --- forEach (Java 8+) ---
        scores.forEach((k, v) -> System.out.println(k + ": " + v));
    }
}

9   LinkedHashMap

import java.util.LinkedHashMap;
import java.util.Map;

public class LinkedHashMapDemo {
    public static void main(String[] args) {

        // --- Insertion-order (default) ---
        Map<String, Integer> map = new LinkedHashMap<>();
        map.put("Banana", 2);
        map.put("Apple", 5);
        map.put("Cherry", 3);

        // Iteration order = insertion order (guaranteed!)
        map.forEach((k, v) -> System.out.println(k + " → " + v));
        // Banana → 2
        // Apple → 5
        // Cherry → 3


        // ========== LRU CACHE (access-order mode) ==========
        int capacity = 3;
        LinkedHashMap<String, String> lruCache = new LinkedHashMap<>(16, 0.75f, true) {
            @Override
            protected boolean removeEldestEntry(Map.Entry<String, String> eldest) {
                return size() > capacity;  // Remove oldest when over capacity
            }
        };

        lruCache.put("A", "1");
        lruCache.put("B", "2");
        lruCache.put("C", "3");
        lruCache.get("A");           // Access A → moves to end
        lruCache.put("D", "4");     // Exceeds capacity → B removed (least recently used)

        System.out.println(lruCache);  // {C=3, A=1, D=4}
    }
}

10   TreeMap

import java.util.TreeMap;
import java.util.NavigableMap;

public class TreeMapDemo {
    public static void main(String[] args) {

        TreeMap<String, Integer> map = new TreeMap<>();

        map.put("Charlie", 3);
        map.put("Alice", 1);
        map.put("Bob", 2);
        map.put("Diana", 4);

        // Keys are automatically sorted!
        System.out.println(map);  // {Alice=1, Bob=2, Charlie=3, Diana=4}

        // --- Navigation ---
        System.out.println(map.firstKey());             // "Alice"
        System.out.println(map.lastKey());              // "Diana"
        System.out.println(map.lowerKey("Charlie"));    // "Bob"    (strictly less)
        System.out.println(map.higherKey("Bob"));       // "Charlie" (strictly greater)

        // --- Sub maps ---
        System.out.println(map.headMap("Charlie"));     // {Alice=1, Bob=2}
        System.out.println(map.tailMap("Charlie"));     // {Charlie=3, Diana=4}
        System.out.println(map.subMap("Bob", "Diana")); // {Bob=2, Charlie=3}

        // --- Descending ---
        System.out.println(map.descendingMap());        // {Diana=4, Charlie=3, Bob=2, Alice=1}

        // --- Poll ---
        System.out.println(map.pollFirstEntry());  // Alice=1 (removed)
        System.out.println(map.pollLastEntry());   // Diana=4 (removed)
        System.out.println(map);                   // {Bob=2, Charlie=3}

        // --- Custom Comparator (case-insensitive) ---
        TreeMap<String, Integer> ciMap = new TreeMap<>(String.CASE_INSENSITIVE_ORDER);
        ciMap.put("apple", 1);
        ciMap.put("Apple", 2);  // Overwrites! (case-insensitive = same key)
        System.out.println(ciMap);  // {apple=2}
    }
}

11   ConcurrentHashMap

import java.util.concurrent.ConcurrentHashMap;
import java.util.Map;

public class ConcurrentHashMapDemo {
    public static void main(String[] args) {

        Map<String, Integer> map = new ConcurrentHashMap<>();

        // Same API as HashMap
        map.put("Java", 1);
        map.put("Python", 2);
        map.put("Go", 3);

        // ❌ null NOT allowed (unlike HashMap)
        // map.put(null, 4);       // throws NullPointerException
        // map.put("Rust", null);  // throws NullPointerException

        // --- Thread-safe atomic operations ---
        map.putIfAbsent("Java", 99);   // No change (already exists)

        // Atomic compute
        map.compute("Java", (key, val) -> val + 10);  // Java → 11

        // Atomic merge (great for counting)
        map.merge("Python", 1, Integer::sum);  // Python → 3

        // --- Safe iteration (no ConcurrentModificationException) ---
        map.forEach((k, v) -> System.out.println(k + ": " + v));

        // --- Multi-threaded usage example ---
        // Multiple threads can read/write simultaneously without external synchronization
        Runnable writer = () -> {
            for (int i = 0; i < 1000; i++) {
                map.merge("counter", 1, Integer::sum);
            }
        };

        Thread t1 = new Thread(writer);
        Thread t2 = new Thread(writer);
        t1.start();
        t2.start();
        try { t1.join(); t2.join(); } catch (InterruptedException e) {}

        System.out.println(map.get("counter"));  // 2000 (always correct, thread-safe)
    }
}

12   EnumMap & EnumSet

import java.util.EnumMap;
import java.util.EnumSet;

public class EnumCollectionsDemo {

    enum Day { MON, TUE, WED, THU, FRI, SAT, SUN }

    public static void main(String[] args) {

        // ========== EnumMap ==========
        EnumMap<Day, String> schedule = new EnumMap<>(Day.class);

        schedule.put(Day.MON, "Gym");
        schedule.put(Day.WED, "Meeting");
        schedule.put(Day.FRI, "Movie");
        schedule.put(Day.SUN, "Rest");

        // Iteration order = enum declaration order (always!)
        schedule.forEach((day, activity) ->
            System.out.println(day + " → " + activity)
        );
        // MON → Gym
        // WED → Meeting
        // FRI → Movie
        // SUN → Rest


        // ========== EnumSet ==========
        EnumSet<Day> weekdays = EnumSet.range(Day.MON, Day.FRI);    // [MON..FRI]
        EnumSet<Day> weekend  = EnumSet.of(Day.SAT, Day.SUN);       // [SAT, SUN]
        EnumSet<Day> allDays  = EnumSet.allOf(Day.class);            // All 7
        EnumSet<Day> noDays   = EnumSet.noneOf(Day.class);           // Empty
        EnumSet<Day> notWeekend = EnumSet.complementOf(weekend);    // [MON..FRI]

        System.out.println("Weekdays: " + weekdays);
        System.out.println("Weekend: " + weekend);

        if (weekdays.contains(Day.MON)) {
            System.out.println("Monday is a weekday");  // O(1) — backed by a bit vector
        }
    }
}

13   Immutable / Unmodifiable Collections (Java 9+)

import java.util.*;

public class ImmutableDemo {
    public static void main(String[] args) {

        // ========== Java 9+ Factory Methods ==========

        // Immutable List
        List<String> list = List.of("A", "B", "C");
        // list.add("D");  → UnsupportedOperationException!

        // Immutable Set
        Set<String> set = Set.of("X", "Y", "Z");
        // Set.of("A", "A");  → IllegalArgumentException! (duplicates not allowed)

        // Immutable Map
        Map<String, Integer> map = Map.of(
            "Alice", 1,
            "Bob", 2,
            "Charlie", 3
        );

        // Immutable Map with more than 10 entries
        Map<String, Integer> bigMap = Map.ofEntries(
            Map.entry("A", 1),
            Map.entry("B", 2),
            Map.entry("C", 3)
        );


        // ========== Unmodifiable Wrappers ==========

        List<String> mutable = new ArrayList<>(List.of("A", "B"));
        List<String> unmod = Collections.unmodifiableList(mutable);
        // unmod.add("C");  → UnsupportedOperationException!
        // BUT: mutable.add("C") still works and affects unmod! (it's just a view)


        // ========== Java 10+ copyOf ==========

        List<String> copy = List.copyOf(mutable);  // True independent copy, immutable
        // Modifying 'mutable' does NOT affect 'copy'
    }
}

14   Sorting & Comparators

import java.util.*;

public class SortingDemo {
    public static void main(String[] args) {

        // ========== Sort a List (natural order) ==========
        List<Integer> nums = new ArrayList<>(List.of(5, 2, 8, 1, 9));
        Collections.sort(nums);              // [1, 2, 5, 8, 9]
        nums.sort(Comparator.reverseOrder()); // [9, 8, 5, 2, 1]


        // ========== Sort Strings ==========
        List<String> names = new ArrayList<>(List.of("Charlie", "Alice", "bob"));
        names.sort(Comparator.naturalOrder());                   // [Alice, Charlie, bob]  (uppercase first)
        names.sort(String.CASE_INSENSITIVE_ORDER);               // [Alice, bob, Charlie]
        names.sort(Comparator.comparing(String::length));        // [bob, Alice, Charlie]  (by length)


        // ========== Sort Custom Objects ==========
        List<Student> students = new ArrayList<>(List.of(
            new Student("Alice", 90),
            new Student("Bob", 95),
            new Student("Charlie", 90)
        ));

        // Sort by score descending, then name ascending
        students.sort(
            Comparator.comparingInt(Student::score).reversed()
                       .thenComparing(Student::name)
        );

        students.forEach(System.out::println);
        // Student[name=Bob, score=95]
        // Student[name=Alice, score=90]
        // Student[name=Charlie, score=90]


        // ========== Comparable (natural ordering on the class itself) ==========
        // If Student implements Comparable<Student>, you can use:
        // Collections.sort(students);  → uses compareTo()
    }

    // Java 16+ record (or a class with getters)
    record Student(String name, int score) {}
}

💡 Comparable vs Comparator

Comparable → class implements compareTo() → defines the natural ordering (one way to sort).
Comparator → external object → defines custom ordering (can have many different sort strategies).

15   Collections Utility Methods

import java.util.*;

public class UtilityDemo {
    public static void main(String[] args) {

        List<Integer> list = new ArrayList<>(List.of(3, 1, 4, 1, 5, 9));

        // --- Sort ---
        Collections.sort(list);                       // [1, 1, 3, 4, 5, 9]

        // --- Reverse ---
        Collections.reverse(list);                    // [9, 5, 4, 3, 1, 1]

        // --- Shuffle ---
        Collections.shuffle(list);                    // Random order

        // --- Min / Max ---
        int min = Collections.min(list);               // 1
        int max = Collections.max(list);               // 9

        // --- Frequency ---
        int count = Collections.frequency(list, 1);   // 2 (how many times 1 appears)

        // --- Fill ---
        Collections.fill(list, 0);                    // [0, 0, 0, 0, 0, 0]

        // --- nCopies ---
        List<String> fiveHellos = Collections.nCopies(5, "Hello");  // [Hello, Hello, Hello, Hello, Hello]

        // --- Singleton (immutable single-element collection) ---
        List<String> single = Collections.singletonList("only");
        Set<String> singleSet = Collections.singleton("only");
        Map<String, Integer> singleMap = Collections.singletonMap("key", 1);

        // --- Synchronized wrappers ---
        List<String> syncList = Collections.synchronizedList(new ArrayList<>());
        Map<String, Integer> syncMap = Collections.synchronizedMap(new HashMap<>());

        // --- Unmodifiable wrappers ---
        List<Integer> readOnly = Collections.unmodifiableList(new ArrayList<>(List.of(1, 2, 3)));

        // --- Swap ---
        List<String> swapList = new ArrayList<>(List.of("A", "B", "C"));
        Collections.swap(swapList, 0, 2);           // [C, B, A]

        // --- Rotate ---
        List<String> rotList = new ArrayList<>(List.of("A", "B", "C", "D"));
        Collections.rotate(rotList, 2);              // [C, D, A, B]

        // --- Binary Search (list must be sorted!) ---
        List<Integer> sorted = List.of(1, 3, 5, 7, 9);
        int index = Collections.binarySearch(sorted, 5);  // 2  (found at index 2)

        // --- Disjoint (no common elements?) ---
        boolean noOverlap = Collections.disjoint(
            List.of(1, 2), List.of(3, 4));  // true
    }
}

16   Streams + Collections

import java.util.*;
import java.util.stream.*;

public class StreamsDemo {
    public static void main(String[] args) {

        List<String> names = List.of("Alice", "Bob", "Charlie", "Anna", "Brian");

        // --- Filter ---
        List<String> aNames = names.stream()
            .filter(n -> n.startsWith("A"))
            .collect(Collectors.toList());              // [Alice, Anna]

        // --- Map (transform) ---
        List<Integer> lengths = names.stream()
            .map(String::length)
            .collect(Collectors.toList());              // [5, 3, 7, 4, 5]

        // --- Collect to Set ---
        Set<String> nameSet = names.stream()
            .collect(Collectors.toSet());

        // --- Collect to Map ---
        Map<String, Integer> nameToLength = names.stream()
            .collect(Collectors.toMap(n -> n, String::length));
        // {Alice=5, Bob=3, Charlie=7, Anna=4, Brian=5}

        // --- Grouping ---
        Map<Integer, List<String>> byLength = names.stream()
            .collect(Collectors.groupingBy(String::length));
        // {3=[Bob], 4=[Anna], 5=[Alice, Brian], 7=[Charlie]}

        // --- Sorting ---
        List<String> sorted = names.stream()
            .sorted()
            .collect(Collectors.toList());              // [Alice, Anna, Bob, Brian, Charlie]

        // --- Reduce ---
        int totalLen = names.stream()
            .mapToInt(String::length)
            .sum();                                     // 24

        // --- Joining ---
        String joined = names.stream()
            .collect(Collectors.joining(", "));         // "Alice, Bob, Charlie, Anna, Brian"

        // --- Distinct + Count ---
        List<Integer> nums = List.of(1, 2, 2, 3, 3, 3);
        long unique = nums.stream().distinct().count();  // 3

        // --- toUnmodifiableList (Java 10+) ---
        List<String> immutable = names.stream()
            .filter(n -> n.length() > 3)
            .collect(Collectors.toUnmodifiableList());  // Immutable result

        // --- Partitioning ---
        Map<Boolean, List<String>> partitioned = names.stream()
            .collect(Collectors.partitioningBy(n -> n.length() > 4));
        // {false=[Bob, Anna], true=[Alice, Charlie, Brian]}

        System.out.println(partitioned);
    }
